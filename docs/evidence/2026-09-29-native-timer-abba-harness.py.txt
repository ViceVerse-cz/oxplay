def verify_variant(meta,variant):

  H.require(H.sha(path)==item['sha256'],'recorded native dependency changed')
 return app,lib

def shared_services(resource):
 output={}
 for name in ('WindowServer','coreaudiod'):
  found=[]
  for sample in resource['samples']:
   rows=[row for row in sample['host_other_top_cpu'] if row['executable']==name]
   H.require(len(rows)<=1,'duplicate shared service in bounded host audit')
   if rows:found.append(rows[0]['cpu_one_core_percent'])
  output[name]={'observed_sample_count':len(found),'omitted_sample_count':60-len(found),'mean_when_present':sum(found)/len(found) if found else None,'peak_when_present':max(found) if found else None,'scope':'Shared host service outside app total; top-eight omission is unknown, not zero. No exclusive app attribution.'}
 return output
